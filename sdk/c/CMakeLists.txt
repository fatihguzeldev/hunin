cmake_minimum_required(VERSION 3.16)
project(hunin_sdk VERSION 0.1.0 LANGUAGES C CXX)
find_package(Threads REQUIRED)
add_library(hunin_sdk src/hunin.c)
set_target_properties(hunin_sdk PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_include_directories(hunin_sdk PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/../cpp/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(hunin_sdk PUBLIC Threads::Threads)
install(TARGETS hunin_sdk ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES include/hunin.h ../cpp/include/hunin.hpp DESTINATION include)

option(HUNIN_SDK_BUILD_TESTS "Build the SDK concurrency contract test" OFF)
if(HUNIN_SDK_BUILD_TESTS)
  enable_testing()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(hunin_sdk_concurrency tests/concurrency.cpp)
  target_compile_features(hunin_sdk_concurrency PRIVATE cxx_std_17)
  target_link_libraries(hunin_sdk_concurrency PRIVATE hunin_sdk)
  add_test(NAME sdk_context_concurrency COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/verify.py $<TARGET_FILE:hunin_sdk_concurrency>)
endif()
