cmake_minimum_required(VERSION 3.16)
project(native_lifetimes LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(phase_a SHARED phase.cpp)
add_library(phase_b SHARED phase.cpp)
target_compile_definitions(phase_a PRIVATE PHASE_SEED=17)
target_compile_definitions(phase_b PRIVATE PHASE_SEED=29)
add_executable(lifetime_target main.cpp)
target_compile_features(lifetime_target PRIVATE cxx_std_17)
target_link_libraries(lifetime_target PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
